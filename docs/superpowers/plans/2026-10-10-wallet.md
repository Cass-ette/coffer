**RawJSON.swift**（动态 JSON，教科书款，无第三方依赖）：

```swift
/// 动态 JSON 值。用于把未知 payload 原样保留在内存并在 encode 时回写。
public enum RawJSON: Codable, Hashable, Sendable {
    case string(String), number(Double), bool(Bool), null
    case array([RawJSON]), object([String: RawJSON])

    public init(from decoder: Decoder) throws {
        let c = try decoder.singleValueContainer()
        if let v = try? c.decode(Bool.self) { self = .bool(v) }
        else if let v = try? c.decode(String.self) { self = .string(v) }
        else if let v = try? c.decode(Double.self) { self = .number(v) }
        else if (try? c.decodeNil()) == true { self = .null }
        else if let v = try? c.decode([RawJSON].self) { self = .array(v) }
        else { self = .object(try c.decode([String: RawJSON].self)) }
    }
    public func encode(to encoder: Encoder) throws {
        var c = encoder.singleValueContainer()
        switch self {
        case .string(let v): try c.encode(v)
        case .number(let v): try c.encode(v)
        case .bool(let v): try c.encode(v)
        case .null: try c.encodeNil()
        case .array(let v): try c.encode(v)
        case .object(let v): try c.encode(v)
        }
    }
    /// RawJSON → Data（经 JSONEncoder 走自己的 encode，object/array 都合法）
    public var jsonData: Data? { try? JSONEncoder().encode(self) }
    /// Data → RawJSON
    public static func from(_ data: Data) -> RawJSON? { try? JSONDecoder().decode(RawJSON.self, from: data) }
}
```

**EntryPayload** 改自定义 Codable。**磁盘真实格式（实测验证）：`{"camelCaseCase名":{"_0":{…}}}`**——case 名是 `apiKey/sshKey/secureNote` 这些 Swift 标识符（不是 EntryType 的 api_key rawValue），关联值包在 `_0` 键下。自定义实现必须**逐字节复刻**这个格式，否则旧 vault 解不出、新文件旧 app 读不了（双输）。

**探针已验证的实现方案**（/tmp/syn-probe 实测通过：_0 解包 OK、unknown 保留 OK、wrap 形状与合成格式一致）：

```swift
public enum EntryPayload: Codable, Hashable, Sendable {
    case login(LoginPayload)
    // ...现有全部 + wallet...
    case unknown(RawJSON)

    public init(from decoder: Decoder) throws {
        let dict = try decoder.singleValueContainer().decode([String: RawJSON].self)
        // 注意：不要写成嵌套泛型函数（guard let + case 模式匹配组合会触发
        // Swift 6.4 编译器 bug "failed to produce diagnostic"，探针实测踩过）。
        // 用平铺 switch + 每 case 内联解包：
        self = .unknown(.object(dict))   // 先兜底，下面命中再覆盖
        if let caseVal = dict["login"],
           case .object(let inner) = caseVal,
           let zero = inner["_0"], let d = zero.jsonData,
           let p = try? JSONDecoder().decode(LoginPayload.self, from: d) {
            self = .login(p); return
        }
        if let caseVal = dict["access"],
           case .object(let inner) = caseVal,
           let zero = inner["_0"], let d = zero.jsonData,
           let p = try? JSONDecoder().decode(AccessPayload.self, from: d) {
            self = .access(p); return
        }
        // ...apiKey("apiKey")、sshKey("sshKey")、database、totp、
        //    secureNote("secureNote")、wallet("wallet") 同款 5 行...
    }

    public func encode(to encoder: Encoder) throws {
        var c = encoder.singleValueContainer()
        switch self {
        case .login(let p): try c.encode(Self.wrap("login", p))
        // ...其余已知 case 同款...
        case .wallet(let p): try c.encode(Self.wrap("wallet", p))
        case .unknown(let raw): try c.encode(raw)   // 原样回写，_0 与否都原封不动
        }
    }

    /// encode 具体类型 → RawJSON → 包 {"case名":{"_0":…}}（复刻合成格式；顶层静态函数不踩编译器 bug）
    private static func wrap<P: Encodable>(_ key: String, _ p: P) -> [String: RawJSON] {
        let raw = RawJSON.from(try! JSONEncoder().encode(p)) ?? .null
        return [key: .object(["_0": raw])]
    }
}
```

关键核对点（写完立刻跑 `testAllSixPayloadsRoundtrip` + 新格式锁测试 testPayloadEncodesSynthesizedShape）：
- 解码键是 `apiKey/sshKey/secureNote`（case 名 camelCase），**不是** `api_key/ssh_key/secure_note`（那是 EntryType 的 rawValue，两套别混）
- `try?` 解包语义：键在但类型对不上（损坏条目）会落到 `.unknown` 保原文，不抛错不丢数据——这是想要的
- 嵌套泛型函数 + guard case 模式匹配 = Swift 6.4 编译器 bug，用平铺 if + 顶层静态 wrap 规避（探针实测）

**EntryType** 加 `case unknown`（放最后）。**Entry** 加：

```swift

```

CodingKeys 加 `degradedTypeRaw`。init(from:) 中 type 解码改：

```swift
import Foundation
import TronKit
import HdWalletKit

/// Tron 纯函数层：地址校验、助记词派生。无网络、无状态，可安全单测。
/// 转账的 secret 解密与 Touch ID 门在 app 层（TransferSheet），本类不碰 vault。
public final class TronWalletService {
    public init() {}

    /// Base58Check 校验 + T 前缀
    public func validateAddress(_ s: String) -> Bool {
        (try? Address(address: s)) != nil && s.hasPrefix("T")
    }

    /// 12/24 词助记词 → Tron 地址（BIP44 m/44'/195'/0'/0/0，钱包标准路径）
    public func address(fromMnemonic mnemonic: String) throws -> String {
        let words = mnemonic.trimmingCharacters(in: .whitespacesAndNewlines)
            .split(separator: " ").map(String.init)
        try Mnemonic.validate(words: words)
        guard let seed = Mnemonic.seed(mnemonic: words) else {
            throw TronWalletError.badMnemonic
        }
        return try Signer.address(seed: seed).base58
    }

    /// 助记词 → 32 字节私钥（hex 字符串返回，存储格式与 WalletPayload.secret 一致）
    public func privateKey(fromMnemonic mnemonic: String) throws -> String {
        let words = mnemonic.trimmingCharacters(in: .whitespacesAndNewlines)
            .split(separator: " ").map(String.init)
        try Mnemonic.validate(words: words)
        guard let seed = Mnemonic.seed(mnemonic: words) else {
            throw TronWalletError.badMnemonic
        }
        let pk = try Signer.privateKey(seed: seed)
        return pk.map { String(format: "%02x", $0) }.joined()
    }

    /// 私钥（64 hex 字符）→ 地址
    public func address(fromPrivateKey privateKeyHex: String) throws -> String {
        let pk = try Signer.privateKey(string: privateKeyHex)
        return try Signer.address(privateKey: pk).base58
    }

    /// 生成 12 词助记词。熵注入用 SecRandom（系统加密随机源，spec §6.1 安全要求）
    public func generateMnemonic(entropy: Data? = nil) throws -> [String] {
        if let entropy { return Mnemonic.generate(entropy: entropy) }
        var buf = Data(count: 16)
        let status = buf.withUnsafeMutableBytes {
            SecRandomCopyBytes(kSecRandomDefault, 16, $0.baseAddress!)
        }
        guard status == errSecSuccess else { throw TronWalletError.randomFailed }
        return Mnemonic.generate(entropy: buf)
    }
}

public enum TronWalletError: Error, Equatable {
    case badMnemonic
    case randomFailed
}
```

注意：`Signer.privateKey(seed:)` / `Mnemonic.generate(entropy:)` / `Mnemonic.validate(words:)` 是冒烟时实际读过的 TronKit/HdWalletKit 公开 API（Signer.swift:46-50、Mnemonic.swift:56/120）。`Signer` init 是 internal，但静态工厂（`.address(seed:)` `.privateKey(seed:)` `.privateKey(string:)`）都 public。

```swift
import Foundation

/// HTTP 注入点：单测 mock，生产 URLSession
public protocol HTTPProviding {
    func post(_ path: String, body: [String: Any], headers: [String: String]) async throws -> Data
    func get(_ path: String, headers: [String: String]) async throws -> Data
}

public final class URLSessionHTTP: HTTPProviding {
    let base: URL
    public init(base: URL = URL(string: "https://api.trongrid.io")!) { self.base = base }

    public func post(_ path: String, body: [String: Any], headers: [String: String]) async throws -> Data {
        var req = URLRequest(url: base.appendingPathComponent(path), timeoutInterval: 10)
        req.httpMethod = "POST"
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")
        headers.forEach { req.setValue($1, forHTTPHeaderField: $0) }
        req.httpBody = try JSONSerialization.data(withJSONObject: body)
        return try await run(req)
    }
    public func get(_ path: String, headers: [String: String]) async throws -> Data {
        var req = URLRequest(url: base.appendingPathComponent(path), timeoutInterval: 10)
        headers.forEach { req.setValue($1, forHTTPHeaderField: $0) }
        return try await run(req)
    }
    private func run(_ req: URLRequest) async throws -> Data {
        let (data, resp) = try await URLSession.shared.data(for: req)
        guard let http = resp as? HTTPURLResponse, (200..<300).contains(http.statusCode) else {
            throw TronGridError.http((resp as? HTTPURLResponse)?.statusCode ?? -1)
        }
        return data
    }
}

public struct TronBalances: Equatable, Sendable {
    public let trxBalance: Int64   // sun（TRX 6 位小数）
    public let usdtSun: Int64      // USDT 最小单位
    public init(trxBalance: Int64, usdtSun: Int64) {
        self.trxBalance = trxBalance; self.usdtSun = usdtSun
    }
}

public enum TronGridError: Error { case http(Int), badResponse }

/// 主网 USDT（TRC-20）合约，spec §5.2 硬编码合理
public let usdtContract = "TR7NHqjeKQxGTCi8q8ZY4pL8otSzgjLj6t"

public final class TronGridClient {
    let http: HTTPProviding
    let headers: [String: String]   // 空 key 时为空字典（免 key 可用但限流紧，spec §5.2）
    public init(http: HTTPProviding, apiKey: String = "") {
        self.http = http
        self.headers = apiKey.isEmpty ? [:] : ["TRON-PRO-API-KEY": apiKey]
    }

    /// /v1/accounts/{addr}：TRX balance + trc20 数组里的 USDT
    public func balances(address: String) async throws -> TronBalances {
        let data = try await http.get("/v1/accounts/\(address)", headers: headers)
        let obj = (try? JSONSerialization.jsonObject(with: data)) as? [String: Any]
        guard obj?["success"] as? Bool == true else { throw TronGridError.badResponse }
        let accounts = obj?["data"] as? [[String: Any]] ?? []
        guard let acc = accounts.first else {
            return TronBalances(trxBalance: 0, usdtSun: 0)  // 未激活账户 = 0
        }
        let trx = acc["balance"] as? Int64 ?? Int64(acc["balance"] as? Int ?? 0)
        var usdt: Int64 = 0
        if let trc20 = acc["trc20"] as? [[String: Any]] {
            for row in trc20 {
                if let v = row[usdtContract] as? String, let n = Int64(v) { usdt = n }
            }
        }
        return TronBalances(trxBalance: trx, usdtSun: usdt)
    }

    /// 用户输入金额（字符串）→ 最小单位 Int64。USDT 6 位小数，禁 Double。
    public static func parseSun(_ input: String) -> Int64? {
        let parts = input.split(separator: ".")
        guard let whole = Int64(parts[0]) else { return nil }
        guard parts.count <= 2 else { return nil }
        var frac = Int64(0); var digits = 0
        if parts.count == 2 {
            let f = String(parts[1])
            guard f.count <= 6, f.allSatisfy(\.isNumber) else { return nil }
            digits = f.count
            frac = Int64(f) ?? 0
        }
        let scale = Int64(truncating: pow(10, 6 - digits) as NSDecimalNumber)
        return whole * 1_000_000 + frac * scale
    }

    /// 展示格式化（唯一出口）：8250355783 → "8250.355783"
    public static func formatUSDT(_ sun: Int64) -> String {
        let neg = sun < 0
        let abs = sun.magnitude
        let whole = abs / 1_000_000, frac = abs % 1_000_000
        return "\(neg ? "-" : "")\(whole).\(String(format: "%06d", frac))"
    }
}
```

- [ ] **Step 3: 实现 TronTransferService**

签名广播只走 TronKit **公开** API（`Signer.sign(hash:privateKey:)`，Signer.swift:54 已核实是 public static；`Signer.signature(hash:)` 是 internal 不能用）。不用 `Kit`（带 GRDB 同步生命周期，过重）。txID 就是待签哈希（TransactionSender.swift:54 佐证：`signer.signature(hash: createdTransaction.txID)`），无需自己算摘要。广播走纯 JSON 路径（TronGridProvider.swift:227 佐证 body 键：txID/raw_data/raw_data_hex/signature/visible）。

```swift
import Foundation
import TronKit

public enum TransferValidationError: Error, Equatable {
    case invalidAddress
    case insufficientUSDT
    case insufficientTRXForFee(neededSun: Int64, haveSun: Int64)
}
public enum BroadcastError: Error { case rejected(String) }

public struct TronFee: Equatable, Sendable {
    public let energy: Int
    public let energyPrice: Int64     // sun/energy（动态查询）
    public let bandwidthBurned: Int   // 免费额度不足时燃烧的点数
    public var feeSun: Int64 { Int64(energy) * energyPrice + Int64(bandwidthBurned) * 1000 }
    public init(energy: Int, energyPrice: Int64, bandwidthBurned: Int) {
        self.energy = energy; self.energyPrice = energyPrice; self.bandwidthBurned = bandwidthBurned
    }
}

public struct TransferReceipt: Equatable, Sendable {
    public let txHash: String
    public var tronscanURL: URL { URL(string: "https://tronscan.org/#/transaction/\(txHash)")! }
    public init(txHash: String) { self.txHash = txHash }
}

public final class TronTransferService {
    let http: HTTPProviding
    public init(http: HTTPProviding) { self.http = http }

    /// 主网 USDT 合约（spec §5.2 常量，注意 spec 文档里旧地址是笔误，以此为准）
    static let usdtContractBase58 = "TR7NHqjeKQxGTCi8q8ZY4pL8otSzgjLj6t"
    static let usdtContractHex = "41a614f803b6fd780986a42c78ec9c7f77e6ded13c"

    /// 转账前校验（地址→余额→费用，顺序即测试断言依据）
    public func validateTransfer(from: String, to: String, usdtSun: Int64) async throws -> TronFee {
        let ws = TronWalletService()
        guard ws.validateAddress(from), ws.validateAddress(to) else { throw TransferValidationError.invalidAddress }
        let bal = try await TronGridClient(http: http).balances(address: from)
        guard bal.usdtSun >= usdtSun else { throw TransferValidationError.insufficientUSDT }
        let fee = try await estimateFee(from: from, to: to, usdtSun: usdtSun)
        guard bal.trxBalance >= fee.feeSun else {
            throw TransferValidationError.insufficientTRXForFee(neededSun: fee.feeSun, haveSun: bal.trxBalance)
        }
        return fee
    }

    public func estimateFee(from: String, to: String, usdtSun: Int64) async throws -> TronFee {
        let simData = try await http.post("/wallet/triggerconstantcontract", body: [
            "owner_address": Self.hex41(from), "contract_address": Self.usdtContractHex,
            "function_selector": "transfer(address,uint256)",
            "parameter": Self.abiEncode(to: to, amount: usdtSun), "visible": false], headers: [:])
        let sim = (try? JSONSerialization.jsonObject(with: simData)) as? [String: Any]
        let energy = sim?["energy_used"] as? Int ?? 64_000

        let paramData = try await http.post("/wallet/getchainparameters", body: [:], headers: [:])
        let pObj = (try? JSONSerialization.jsonObject(with: paramData)) as? [String: Any]
        var energyPrice: Int64 = 420
        for p in (pObj?["chainParameter"] as? [[String: Any]]) ?? [] {
            if p["key"] as? String == "getEnergyFee" {
                energyPrice = (p["value"] as? NSNumber)?.int64Value ?? 420
            }
        }

        let acctData = try await http.get("/v1/accounts/\(from)")
        let aObj = (try? JSONSerialization.jsonObject(with: acctData)) as? [String: Any]
        let accounts = aObj?["data"] as? [[String: Any]] ?? []
        let acc = accounts.first
        let freeLimit = (acc?["freeNetLimit"] as? NSNumber)?.intValue ?? 600
        let freeUsed = (acc?["freeNetUsed"] as? NSNumber)?.intValue ?? 0
        let burn = max(0, 345 - max(0, freeLimit - freeUsed))   // 一笔 TRC-20 ≈ 345 带宽点

        return TronFee(energy: energy, energyPrice: energyPrice, bandwidthBurned: burn)
    }

    /// 构建并签名 USDT transfer，广播。返回受理哈希（受理≠成功，spec §5.4）
    public func sendUSDT(from: String, to: String, usdtSun: Int64,
                         privateKeyHex: String, feeLimitSun: Int64) async throws -> TransferReceipt {
        // 1) 构建原始交易（带 feeLimit——费用上限保护，双重防线）
        let buildData = try await http.post("/wallet/triggersmartcontract", body: [
            "owner_address": Self.hex41(from), "contract_address": Self.usdtContractHex,
            "function_selector": "transfer(address,uint256)",
            "parameter": Self.abiEncode(to: to, amount: usdtSun),
            "fee_limit": feeLimitSun, "visible": false], headers: [:])
        guard let build = (try? JSONSerialization.jsonObject(with: buildData)) as? [String: Any],
              let tx = build["transaction"] as? [String: Any],
              let txID = tx["txID"] as? String else { throw TronGridError.badResponse }

        // 2) 签名：txID 即待签哈希（TronKit TransactionSender 同款做法），public API
        let signature = try Signer.sign(hash: Self.hexData(txID),
                                        privateKey: Self.hexData(privateKeyHex))

        // 3) 广播：原始 transaction 字典 + signature 注入（纯 JSON，无需 protobuf）
        var broadcastBody = tx
        broadcastBody["signature"] = [Self.hexString(signature)]
        let bData = try await http.post("/wallet/broadcasttransaction", body: broadcastBody, headers: [:])
        guard let bObj = (try? JSONSerialization.jsonObject(with: bData)) as? [String: Any] else {
            throw TronGridError.badResponse
        }
        if let r = bObj["result"], !(r is Bool) {   // "DUPLICATE_INPUT" 等即时失败码
            throw BroadcastError.rejected("\(r)")
        }
        return TransferReceipt(txHash: txID)
    }

    // MARK: - 编码工具（有单测锁行为，见下）

    /// transfer(address,uint256) ABI：addr 20 字节 + amount，各补齐 32 字节，hex 无 0x
    static func abiEncode(to: String, amount: Int64) -> String {
        let addrRaw: String
        if let a = try? Address(address: to) {
            addrRaw = a.raw.dropFirst().map { String(format: "%02x", $0) }.joined()  // 去 0x41 前 20 字节
        } else { addrRaw = "" }
        // ⚠️ 实测：String(format: "%064x", Int64) 对 ≥2^32 的值会截断成全零（金额 ≥4295 USDT 就废）
        // 用 String(radix:) 补零，禁用 %x/%llx（%llx 虽实测也对，但 radix 写法无格式化坑）
        let amtHex = String(amount, radix: 16)
        let amt = String(repeating: "0", count: 64 - amtHex.count) + amtHex
        return String(repeating: "0", count: 24) + addrRaw + amt
    }
    /// base58 地址 → 41 前缀 hex（owner_address 用）
    static func hex41(_ base58: String) -> String {
        guard let a = try? Address(address: base58) else { return "" }
        return a.raw.map { String(format: "%02x", $0) }.joined()
    }
    static func hexData(_ s: String) -> Data {
        var d = Data(); var i = s.startIndex
        while i < s.endIndex {
            let j = s.index(i, offsetBy: 2, limitedBy: s.endIndex) ?? s.endIndex
            if let b = UInt8(String(s[i..<j]), radix: 16) { d.append(b) }
            i = j
        }
        return d
    }
    static func hexString(_ d: Data) -> String { d.map { String(format: "%02x", $0) }.joined() }
}
```

**实现注意（执行者必读）：**
① 上面是**骨架代码**：`freeNetLimit`/`freeNetUsed` 两行的 `??` 优先级要显式括号（`(a as? NSNumber)?.intValue ?? 600` 与减法组合易错，写成两行局部变量最稳）。`abiEncode`/`hex41` 用 `String(radix:)` 补零——**禁用 `String(format: "%064x")`**（实测对 ≥2^32 的 Int64 截断成全零，金额超 4295 USDT 就静默转错账）。
② 编码工具补单测锁行为：黄金地址 → `41` 开头 hex；`abiEncode(黄金地址, 1_000_000)` = 24 个 `0` + 地址 20 字节 hex + `…0000000000f4240`（1e6 hex）；`abiEncode(_, 8_589_934_592)` 尾段含 `200000000`（防截断回归）。
③ 广播响应 `result` 字段成功是 `true`（Bool），失败是错误码字符串——用 `!(r is Bool)` 区分。
④ feeLimit 传 `fee.feeSun`（validateTransfer 的返回值由调用方透传），构建请求带 `fee_limit` 字段（费用上限双重防线）。
